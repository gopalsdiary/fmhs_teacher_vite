teacher_database টেবিলের সবাইকে ম্যাসেজ পাঠাতে পারবে। (no photo/video; only text)

message table এর জন্য sql লিখে দাও। 

-- ১. পুরোনো টেবিলগুলো মুছে ফেলা
DROP TABLE IF EXISTS public.fmhs_teacher_message;
DROP TABLE IF EXISTS public.fmhs_teacher_messages;
DROP TABLE IF EXISTS public.teacher_presence;
-- ২. নতুন করে টেবিল তৈরি (সঠিক নামে)
CREATE TABLE public.fmhs_teacher_messages (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  sender_email TEXT NOT NULL,
  recipient_email TEXT NOT NULL,
  message TEXT NOT NULL,
  is_read BOOLEAN DEFAULT false,
  read_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT now()
);
-- ৩. প্রেজেন্স টেবিল তৈরি
CREATE TABLE public.teacher_presence (
    email TEXT PRIMARY KEY,
    last_seen TIMESTAMPTZ DEFAULT NOW(),
    status TEXT DEFAULT 'online'
);
