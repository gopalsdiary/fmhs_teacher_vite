"teacher_email_id"  [https://rtfefxghfbtirfnlbucb.supabase.co/rest/v1/teacher_database]  এর সাথে লগইন ইমেইল মিললে তাকে ম্যাসেজিং করতে দিবে। 
teacher_database টেবিলের সবাইকে ম্যাসেজ পাঠাতে পারবে। (no photo/video; only text)

message table এর জন্য sql লিখে দাও। 

-- আগের টেবিল মুছে ফেলা (সাবধানে রান করবেন)
DROP TABLE IF EXISTS messages;

-- আধুনিক ম্যাসেজ টেবিল (is_read এবং read_at সহ)
CREATE TABLE messages (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    sender_email TEXT NOT NULL,
    recipient_email TEXT NOT NULL,
    message TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    is_read BOOLEAN DEFAULT FALSE,
    read_at TIMESTAMPTZ
);

-- অনলাইন স্ট্যাটাস ট্র্যাকিং টেবিল
CREATE TABLE teacher_presence (
    email TEXT PRIMARY KEY,
    last_seen TIMESTAMPTZ DEFAULT NOW(),
    status TEXT DEFAULT 'online'
);

-- RLS পলিসি সেটআপ
ALTER TABLE messages ENABLE ROW LEVEL SECURITY;
ALTER TABLE teacher_presence ENABLE ROW LEVEL SECURITY;

-- ম্যাসেজ পলিসি
CREATE POLICY "View messages" ON messages FOR SELECT USING (auth.jwt() ->> 'email' = sender_email OR auth.jwt() ->> 'email' = recipient_email);
CREATE POLICY "Insert messages" ON messages FOR INSERT WITH CHECK (auth.jwt() ->> 'email' = sender_email);
CREATE POLICY "Update read status" ON messages FOR UPDATE USING (auth.jwt() ->> 'email' = recipient_email);

-- প্রেজেন্স পলিসি
CREATE POLICY "View presence" ON teacher_presence FOR SELECT USING (true);
CREATE POLICY "Update own presence" ON teacher_presence FOR ALL USING (auth.jwt() ->> 'email' = email);
