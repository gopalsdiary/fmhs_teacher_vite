  student_iid BIGINT NOT NULL,

  message TEXT NOT NULL,
  sender_type TEXT CHECK (sender_type IN ('student', 'teacher')),
  is_read BOOLEAN DEFAULT false,
  read_at TIMESTAMPTZ,
